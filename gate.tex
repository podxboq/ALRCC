\subsection{Puertas cuánticas}
Una puerta de un qubit $\gate{A}$ puede aplicarse sobre el primer qubit de un ket $\ket{a_1 a_2}$ y el resultado es $\gate{A}\ket{a_1}\otimes \ket{a_2}$.
También puede aplicarse sobre el último de los dos qubits y entonces el resultado es $\ket{a_1}\otimes \gate{A}\ket{a_2}$.

\begin{notation}
	Sea $\gate{A}$ una puerta de un qubit y $\ket{a}=\ket{a_1\cdots a_n}$ un ket con $n$ qubits; se usa la siguiente notación equivalente:
	\[
		\bigotimes_{k=1}^n\gate{A}^{\delta_{hk}}\ket{a_k}=
		\begin{cases}
			\ket{a_1\cdots a_h\gate{A}a_{h+1}\cdots a_n}\\
			\ket{a}\gate{A}_h\\
			\stateket{a}{h}{\gate{A}}
		\end{cases}
		,
	\]
	donde $h\in\N_n$.
\end{notation}

\begin{remark}
	A lo largo de este trabajo, las puertas se escriben en forma \emph{postfija}: la
	expresión $\ket{a}\gate{A}_h$ denota la aplicación del operador
	$\gate{A}_h$ al estado $\ket{a}$. En consecuencia, en un producto de puertas
	el factor más a la izquierda se aplica primero, por lo que las expresiones se leen de izquierda
	a derecha en el mismo orden que el circuito correspondiente. Esta convención
	es puramente notacional: $\ket{a}\gate{A}_h\gate{B}_l$ representa
	$\gate{B}_l(\gate{A}_h\ket{a})$ en la notación estándar de operadores.
\end{remark}
Por ejemplo, si $\ket{a}=\ket{a_1 a_2}$ entonces
\[
	\gate{A}\ket{a_1}\otimes \ket{a_2} =
	\begin{cases}
		\ket{a_1\gate{A}a_2}\\
		\ket{a}\gate{A}_1\\
		\stateket{a}{1}{\gate{A}}
	\end{cases}.
\]

\begin{proposition}
	Sea $\gate{A}$ una puerta de un qubit.
	Entonces, para cualesquiera dos índices distintos $h,k \in \mathbb{N}_n$,
	\begin{equation}
		\label{eq:conmutacion-gates}
		\gate{A}_k\gate{A}_{h} = \gate{A}_{h}\gate{A}_k.
	\end{equation}

\end{proposition}
\begin{proof}
	Sea $\ket{a}$ un ket.
	Para cualesquiera dos índices distintos $h,k \in \mathbb{N}_n$:
	\begin{align*}
		\ket{a}\gate{A}_k\gate{A}_{h}
		&=\stateket{a}{k}{\gate{A}}\gate{A}_{h}=\stateket{a}{kh}{\gate{AA}}=\stateket{a}{h}{\gate{A}}\gate{A}_{k}=\ket{a}\gate{A}_{h}\gate{A}_k.
	\end{align*}
\end{proof}

Por la igualdad~(\ref{eq:conmutacion-gates}), la siguiente notación está bien definida.
\begin{notation}
	Sea $\Gamma=\set{k_1,\dots,k_l}\subseteq\mathbb{N}_n$ un conjunto.
	Sea $\gate{A}$ una puerta de un qubit; se usa la siguiente notación:
	\begin{align}
		\gate{A}_{\Gamma} &= \gate{A}_{k_1,\dots,k_l}=\prod_{k\in\Gamma}\gate{A}_k
	\end{align}
\end{notation}

Nótese que, de acuerdo con la notación anterior, para el caso particular del conjunto vacío $\Gamma=\emptyset$, la puerta cuántica resultante es la puerta identidad $\gate{A}_\emptyset=\gate{I}$.

Por ejemplo, si $\ket{a}=\ket{a_1 a_2}$ y $\Gamma = \set{1,2}$, entonces
\[
	\gate{A}\ket{a_1}\otimes \gate{A}\ket{a_2} =
	\begin{cases}
		\ket{a_1\gate{A}a_2\gate{A}}\\
		\ket{a}\gate{A}_{1,2}=\ket{a}\gate{A}_{\Gamma}\\
		\stateket{a}{1,2}{\gate{A}}=\stateket{a}{\Gamma}{\gate{A}}
	\end{cases}.
\]
