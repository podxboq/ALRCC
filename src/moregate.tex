\subsection{Más puertas cuánticas}
Dos puertas de un qubit $\gate{A}$ y $\gate{B}$ pueden aplicarse sobre el primer y el segundo qubit respectivamente de un ket $\ket{a_1 a_2}$ y el resultado es $\gate{A}\ket{a_1}\otimes \gate{B}\ket{a_2}$.

\begin{notation}
	Sea $\gate{A}$ una puerta de un qubit y $\ket{a}=\ket{a_1\cdots a_n}$ un ket con $n$ qubits; se usa la siguiente notación equivalente:
	\[
		\bigotimes_{k=1}^n\gate{A}^{\delta_{hk}}\gate{B}^{\delta_{lk}}\ket{a_k}=
		\begin{cases}
			\ket{a_1\cdots a_h\gate{A}a_{h+1}\cdots a_l\gate{B}a_{l+1}\cdots a_n}\\
			\gate{A}_h\gate{B}_l\ket{a}\\
			\stateket{a}{h,l}{\gate{A}\gate{B}}
		\end{cases}
		,
	\]
	donde $h,l\in\N_n$.
\end{notation}
Por ejemplo, si $\ket{a}=\ket{a_1 a_2}$ entonces
\[
	\gate{A}\ket{a_1}\otimes \gate{B}\ket{a_2} =
	\begin{cases}
		\ket{a_1\gate{A}a_2\gate{B}}\\
		\gate{A}_1\gate{B}_2\ket{a}\\
		\stateket{a}{1,2}{\gate{A}\gate{B}}
	\end{cases}.
\]

\begin{proposition}
	Sean $\gate{A}$ y $\gate{B}$ dos puertas de un qubit.
	Entonces, para cualesquiera dos índices distintos $h,k \in \mathbb{N}_n$,
	\begin{equation}
		\label{eq:conmutacion-two-gates}
		\gate{A}_k\gate{B}_{h} = \gate{B}_{h}\gate{A}_k.
	\end{equation}

\end{proposition}
\begin{proof}
	Sea $\ket{a}$ un ket.
	Para cualesquiera dos índices distintos $h,k \in \mathbb{N}_n$:
	\begin{align*}
		\gate{A}_k\gate{B}_{h}\ket{a}
		&=\gate{A}_{k}\stateket{a}{h}{\gate{B}}=\stateket{a}{kh}{\gate{AB}}=\gate{B}_{h}\stateket{a}{k}{\gate{A}}=\gate{B}_{h}\gate{A}_k\ket{a}.
	\end{align*}
\end{proof}
