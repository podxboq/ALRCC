\subsection{Puertas cuánticas controladas}
Las puertas cuánticas controladas se aplican sobre kets formados por múltiples qubits de modo que uno o más qubits actúan como controladores de una operación específica sobre uno o más qubits del ket.

La puerta controlada más simple $\gate{A}$ sobre un ket $\ket{a}=\ket{a_1\dots a_n}$ tiene un único qubit de control $\ket{a_h}$ y un único qubit objetivo $\ket{a_k}$.

\begin{notation}
	Sea $\ket{a}=\ket{a_1\cdots a_n}$ un ket.
	Sea $\gate{A}$ una puerta cuántica de un qubit.
	La puerta cuántica controlada con qubit de control $\ket{a_h}$ y qubit objetivo $\ket{a_k}$ se denota por $\stategate{A}{h}{k}$:
	\begin{equation}
	\begin{split}
		\label{eq:generic-controlled-gate}
		\ket{a}\stategate{A}{h}{k}&=a_{h0}\stateket{a}{h}{0}+a_{h1} \stateket{a}{h}{1}\stategate{A}{}{k}= \\
		&=a_{h0}\stateket{a}{h}{0}+a_{h1} \stateket{a}{hk}{1\gate{A}}.
	\end{split}
\end{equation}
\end{notation}

Podemos expresar también la puerta cuántica controlada $\stategate{A}{h}{k}$ como
\begin{align*}
	\ket{a}\stategate{A}{h}{k}&=a_{h0}\stateket{a}{h}{0}+a_{h1} \stateket{a}{hk}{1\gate{A}}= \\
	&=a_{h0}\stateket{a}{h}{0}+a_{h1} \stateket{a}{hk}{1\gate{A}}+a_{h1}\stateket{a}{h}{1}-a_{h1} \stateket{a}{h}{1}=\\
	&=\ket{a}-a_{h1} \stateket{a}{h}{1}+a_{h1} \stateket{a}{hk}{1\gate{A}}
\end{align*}

\begin{proposition}
	Sea $\gate{A}$ una puerta de un qubit.
	Entonces, para cualesquiera tres índices distintos $h,k,k^\prime \in \mathbb{N}_n$,
	\begin{align}
		\stategate{A}{h}{k}\stategate{A}{h}{k^\prime} &= \stategate{A}{h}{k^\prime}\stategate{A}{h}{k}
	\end{align}
\end{proposition}

\begin{proof}
	Sea $\ket{a}$ un ket. Para cualesquiera tres índices distintos $h,k,k^\prime \in \mathbb{N}_n$:
	\begin{equation}
		\label{eq:conmutacion-cgates}
		\begin{split}
			\ket{a}\stategate{A}{h}{k}\stategate{A}{h}{k^\prime}
			&=a_{h0}\stateket{a}{h}{0}\stategate{A}{h}{k^\prime}+a_{h1}\stateket{a}{hk}{1\gate{A}}\stategate{A}{h}{k^\prime}=\\
			&=a_{h0}\stateket{a}{h}{0}+a_{h1}\stateket{a}{hkk^\prime}{1\gate{AA}}=\\
			&=a_{h0}\stateket{a}{h}{0}\stategate{A}{h}{k}+a_{h1}\stateket{a}{hk^\prime}{1\gate{A}}\stategate{A}{h}{k}=\\
			&=\ket{a}\stategate{A}{h}{k^\prime}\stategate{A}{h}{k}
		\end{split}
	\end{equation}
\end{proof}

Por la demostración de (\ref{eq:conmutacion-cgates}), la siguiente notación está bien definida.
\begin{notation}
	Sea $\ket{a}=\ket{a_1\cdots a_n}$ un ket.
	Sea $\gate{A}$ una puerta cuántica de un qubit.
	La puerta cuántica controlada con varios qubits de control $\Gamma=\set{h_1,\dots,h_l}\subset\mathbb{N}_n$ y varios qubits objetivo $\Lambda=\set{k_1,\dots,k_m}\subset\mathbb{N}_n$ con $\Gamma\cap\Lambda=\emptyset$ se denota por:
	\begin{align*}
	\ket{a}\gate{A}^{\Gamma}_{\Lambda}&=\ket{a}\stategate{A}{h_1,\dots,h_l}{k_1,\dots, k_m}=a_{h_{1}0}\stateket{a}{h_1}{0}+a_{h_{1}1}a_{h_{2}0}\stateket{a}{h_{1}h_{2}}{10}+\dots+\\
	&+a_{h_{1}1}a_{h_{2}1}\cdots a_{h_{l}0}\stateket{a}{h_{1}h_{2}\dots h_{l-1}h_{l}}{11\dots 10}+
	a_{h_{1}1}a_{h_{2}1}\cdots a_{h_{l}1}\stateket{a}{h_{1}h_{2}\dots h_{l}k_1\dots k_m}{11\dots 1\gate{A}\dots \gate{A}}=\\
	&=\ket{a}-a_{h_{1}1}a_{h_{2}1}\cdots a_{h_{l}1}\stateket{a}{\Gamma}{1}+a_{h_{1}1}a_{h_{2}1}\cdots a_{h_{l}1}\stateket{a}{\Gamma\Lambda}{1\gate{A}}.
	\end{align*}
\end{notation}
Nótese que, de acuerdo con la notación anterior, para el caso particular del conjunto vacío $\Gamma=\emptyset$, la puerta cuántica resultante es $\stategate{A}{\Gamma}{\Lambda}=\gate{A}_{\Lambda}$.
